    public static void main(String[] args) {
        String str = "Hello";
        
        for (int i = str.length() - 1; i >= 0; i--) {
            System.out.print(str.charAt(i));
        }
    }
}

import java.util.Scanner;

public class Login {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String loginId;
        String password;

        while (true) {
            System.out.print("Enter Login ID: ");
            loginId = sc.next();

            System.out.print("Enter Password: ");
            password = sc.next();

            if (loginId.equals("Prasunamba") && password.equals("4321")) {
                System.out.println("Welcome!");

            }
        }

        sc.close();
    }
}

import java.util.Scanner;

public class LoginDoWhile {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String loginId, password;

        do {
            System.out.print("Enter Login ID: ");
            loginId = sc.next();

            System.out.print("Enter Password: ");
            password = sc.next();

            if (loginId.equals("Prasunamba") && password.equals("4321")) {
                System.out.println("Welcome!");
            } else {
                System.out.println("Invalid Login ID or Password. Try Again.");
            }

        } while (!loginId.equals("Prasunamba") || !password.equals("4321"));

        sc.close();
    }
}

public class IfDemo {
    public static void main(String[] args) {
        int age = 20;
        int marks = 75;

        // if
        if (age >= 18)
            System.out.println("Eligible to vote");

        // else if
        if (marks >= 90)
            System.out.println("Grade A");
        else if (marks >= 60)
            System.out.println("Grade B");
        else
            System.out.println("Grade C");

        // Nested if
        if (age >= 18) {
            if (marks >= 60)
                System.out.println("Eligible for interview");
        }
    }
}

import java.util.Scanner;

public class CustomerNames {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        String[] names = new String[5];

        System.out.println("Enter 5 customer names:");
        for (int i = 0; i < names.length; i++) {
            names[i] = sc.nextLine();
        }

        System.out.println("\nCustomer Names:");
        for (String name : names) {
            System.out.println(name);
        }

        sc.close();
    }
}

Use for loop to show the diff

public class BreakContinue {
    public static void main(String[] args) {

        System.out.println("Using break:");
        for (int i = 1; i <= 5; i++) {
            if (i == 3)
                break;
            System.out.println(i);
        }

        System.out.println("\nUsing continue:");
        for (int i = 1; i <= 5; i++) {
            if (i == 3)
                continue;
            System.out.println(i);
        }
    }
}
